int cnt = 0 ;   

void func(int idx,int not_allowed,int mx ,vector<int> &v){
    if(idx==v.size()){
        cnt++ ; 
        return ; 
    }
    // include
    if(v[idx]>=mx){
        func(idx+1,not_allowed,v[idx],v) ;
    }
    else{
        if(v[idx]>not_allowed){
            func(idx+1,max(v[idx]-1,not_allowed),mx,v) ;  
        }
    }
    
    // exclude
    func(idx+1,not_allowed,mx,v) ; 
}
void solve(){
    cnt = 0 ;
    int n ; cin >> n ; 
    vector<int> v(n) ; read(v) ;  
    func(0,0,0,v) ;  
    put(cnt) ; 
}

int func(int idx,int not_allowed,int mx ){
    if(idx==n){
        return 1LL ;  
    }
    if(vis[idx][not_allowed][mx]==tc){
        return dp[idx][not_allowed][mx] ;  
    }
    int tot = 0 ; 
    // include
    int res ; 
    if(v[idx]>=mx){
        // tot = tot + func(idx+1,not_allowed,v[idx]) ;
        res = func(idx+1,not_allowed,v[idx]) ; 
        addmod(tot,res) ; 
    }
    else{
        if(v[idx]>not_allowed){
            // tot = tot + func(idx+1,max(v[idx]-1,not_allowed),mx) ; 
            res = func(idx+1,max(v[idx]-1,not_allowed),mx) ; 
            addmod(tot,res) ; 
        }
    }
    
    // exclude
    // tot = tot + func(idx+1,not_allowed,mx) ;
    res = func(idx+1,not_allowed,mx) ; 
    addmod(tot,res ) ; 
    
    vis[idx][not_allowed][mx] = tc ;  
    return dp[idx][not_allowed][mx] = tot ; 
}
